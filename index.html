1. Total, Average — Command-Line Args & Static Members

P1

Aim: Find total and average using command-line args and static members.

class Calculator {
    static int count = 0;
    static double findTotal(double a, double b) { count++; return a + b; }
    static double findAvg(double a, double b) { return (a + b) / 2.0; }

    public static void main(String[] args) {
        double a = Double.parseDouble(args[0]);
        double b = Double.parseDouble(args[1]);
        System.out.println("Total: " + findTotal(a, b));
        System.out.println("Average: " + findAvg(a, b));
        System.out.println("Called " + count + " time(s)");
    }
}

Output:

java Calculator 10 20
Total: 30.0
Average: 15.0
Called 1 time(s)

3. Method & Constructor Overloading

P3

Aim: Illustrate method and constructor overloading (compile-time polymorphism).

class Overload {
    int a, b;
    Overload()             { a = 0; b = 0; }
    Overload(int x)        { a = x; b = x; }
    Overload(int x, int y) { a = x; b = y; }

    int add(int x, int y)          { return x + y; }
    double add(double x, double y) { return x + y; }
    int add(int x, int y, int z)   { return x + y + z; }

    public static void main(String[] args) {
        Overload o = new Overload(3, 7);
        System.out.println("a=" + o.a + " b=" + o.b);
        System.out.println("add(2,3)=" + o.add(2, 3));
        System.out.println("add(2.5,3.5)=" + o.add(2.5, 3.5));
        System.out.println("add(1,2,3)=" + o.add(1, 2, 3));
    }
}

Output:

a=3 b=7
add(2,3)=5
add(2.5,3.5)=6.0
add(1,2,3)=6

4. Parameter Passing Using Objects

P4

Aim: Demonstrate passing objects as parameters.

class Box {
    double l, w, h;
    Box(double l, double w, double h) { this.l=l; this.w=w; this.h=h; }
    double volume() { return l * w * h; }
    boolean isEqual(Box b) { return (l==b.l && w==b.w && h==b.h); }

    public static void main(String[] args) {
        Box b1 = new Box(10,20,30), b2 = new Box(10,20,30);
        Box b3 = new Box(5,10,15);
        System.out.println("Volume: " + b1.volume());
        System.out.println("b1==b2? " + b1.isEqual(b2));
        System.out.println("b1==b3? " + b1.isEqual(b3));
    }
}

Output:

Volume: 6000.0
b1==b2? true
b1==b3? false

5. Array Manipulation

P5

Aim: Array sorting (Bubble Sort) and searching (Linear Search).

class ArrayDemo {
    public static void main(String[] args) {
        int[] a = {64, 25, 12, 22, 11};
        for (int i = 0; i < a.length-1; i++)
            for (int j = 0; j < a.length-i-1; j++)
                if (a[j] > a[j+1]) { int t=a[j]; a[j]=a[j+1]; a[j+1]=t; }
        System.out.print("Sorted: ");
        for (int x : a) System.out.print(x + " ");
        int key = 22;
        for (int i = 0; i < a.length; i++)
            if (a[i] == key) System.out.println("\n" + key + " at index " + i);
    }
}

Output:

Sorted: 11 12 22 25 64
22 at index 3

1. Types of Inheritance

P6

class Animal { void eat() { System.out.println("Animal eats"); } }
class Dog extends Animal { void bark() { System.out.println("Dog barks"); } }
class Puppy extends Dog { void weep() { System.out.println("Puppy weeps"); } }

        Athlete a = new Athlete(); a.swim(); a.run();
    }
}

Dog barks
Puppy weeps
Athlete swims
Athlete runs

2. Method Overriding

P7

Aim: Subclass overrides parent method — runtime polymorphism.

class Shape { void area() { System.out.println("Area of shape"); } }
class Circle extends Shape {
    double r;
    Circle(double r) { this.r = r; }
    void area() { System.out.println("Circle: " + (3.14*r*r)); }
}
class Rect extends Shape {
    double l, b;
    Rect(double l, double b) { this.l=l; this.b=b; }
    void area() { System.out.println("Rect: " + (l*b)); }
}
class OverrideDemo {
    public static void main(String[] args) {
        Shape s = new Circle(5);  s.area();
        s = new Rect(4, 6);      s.area();
    }
}

Output:

Circle: 78.5
Rect: 24.0

2. Exception Handling — Multiple Catch Blocks

P11

Aim: Handle different exceptions with multiple catch blocks + finally.

class MultiCatch {
    public static void main(String[] args) {
        try {
            int[] arr = {1, 2, 3};
            System.out.println(arr[5]);
            int x = 10 / 0;
        } catch (ArrayIndexOutOfBoundsException e) {
            System.out.println("Array error: " + e);
        } catch (ArithmeticException e) {
            System.out.println("Math error: " + e);

            System.out.println("Error: " + e);

}

2. Thread Priorities

P14

Aim: Set thread priorities (1=MIN, 5=NORM, 10=MAX). Higher = more CPU time.

class PThread extends Thread {
    PThread(String s) { super(s); }
    public void run() {
        System.out.println(getName() + " pri=" + getPriority());
        for (int i = 1; i <= 3; i++) System.out.println(getName() + "-" + i);
    }
    public static void main(String[] args) {
        PThread t1=new PThread("LOW"), t2=new PThread("MID"), t3=new PThread("HIGH");
        t1.setPriority(Thread.MIN_PRIORITY);
        t2.setPriority(Thread.NORM_PRIORITY);
        t3.setPriority(Thread.MAX_PRIORITY);
        t1.start(); t2.start(); t3.start();
    }
}

Output:

HIGH pri=10
HIGH-1, 2, 3 (generally runs first)
MID pri=5 | LOW pri=1

3. Synchronization

P15

Aim: synchronized keyword locks method — one thread at a time.

class PrintTable {
    synchronized void print(int n) {
        for (int i = 1; i <= 5; i++) {
            System.out.println(Thread.currentThread().getName()
                + ": " + n + "x" + i + "=" + (n*i));
            try { Thread.sleep(200); } catch (Exception e) {}
        }
    }
}
class SyncDemo {
    public static void main(String[] args) {
        PrintTable pt = new PrintTable();
        new Thread(() -> pt.print(5), "T1").start();
        new Thread(() -> pt.print(10), "T2").start();
    }
}

Output:

T1: 5x1=5 ... 5x5=25 (completes first)
T2: 10x1=10 ... 10x5=50

2. File Read & Write (Line by Line)

P17

Aim: Read from one file, write to another using BufferedReader/Writer.

import java.io.*;
class FileCopy {
    public static void main(String[] args) throws IOException {
        BufferedWriter bw = new BufferedWriter(new FileWriter("input.txt"));
        bw.write("Hello line 1\nThis is line 2\nEnd");
        bw.close();

        BufferedReader br = new BufferedReader(new FileReader("input.txt"));
        BufferedWriter out = new BufferedWriter(new FileWriter("output.txt"));
        String line;
        while ((line = br.readLine()) != null) {
            System.out.println("Read: " + line);
            out.write(line + "\n");
        }
        br.close(); out.close();
        System.out.println("Copied to output.txt!");
    }
}

Output:

Read: Hello line 1
Read: This is line 2
Read: End
Copied to output.txt!

1. Event Handling (Keyboard & Mouse)

P18

Aim: Handle key presses & mouse clicks using AWT listeners.

import java.awt.*;
import java.awt.event.*;
class EventDemo extends Frame implements KeyListener, MouseListener {
    String msg = "Press key or click mouse";
    EventDemo() {
        setSize(400,300); setTitle("Events"); setVisible(true);
        addKeyListener(this); addMouseListener(this);
        addWindowListener(new WindowAdapter() {
            public void windowClosing(WindowEvent e) { System.exit(0); }
        });
    }
    public void keyPressed(KeyEvent e) {
        msg = "Key: " + e.getKeyChar(); repaint();
    }
    public void keyReleased(KeyEvent e) {}
    public void keyTyped(KeyEvent e) {}
    public void mouseClicked(MouseEvent e) {
        msg = "Click (" + e.getX() + "," + e.getY() + ")"; repaint();
    }
    public void mousePressed(MouseEvent e) {}
    public void mouseReleased(MouseEvent e) {}
    public void mouseEntered(MouseEvent e) {}
    public void mouseExited(MouseEvent e) {}
    public void paint(Graphics g) { g.drawString(msg, 50, 150); }
    public static void main(String[] args) { new EventDemo(); }
}

Output:

GUI window opens
Press key → shows character
Click mouse → shows (x,y)
